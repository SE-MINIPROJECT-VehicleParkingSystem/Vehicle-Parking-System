const ParkingSlot = require('../models/ParkingSlot');
const ParkingTransaction = require('../models/ParkingTransaction');
const { sendSuccess } = require('../utils/apiResponse');

async function getDashboard(req, res, next) {
  try {
    // Count slots by their saved status.
    const [
      totalSlots,
      availableSlots,
      occupiedSlots,
      reservedSlots,
      activeSessions,
      completedSessions,
      recentTransactions,
      slots,
    ] = await Promise.all([
      ParkingSlot.countDocuments(),
      ParkingSlot.countDocuments({ status: 'Available' }),
      ParkingSlot.countDocuments({ status: 'Occupied' }),
      ParkingSlot.countDocuments({ status: 'Reserved' }),

      // Vehicles that have entered but have not exited yet.
      ParkingTransaction.countDocuments({ exitTime: null }),

      // Sessions that have an exit time.
      ParkingTransaction.countDocuments({
        exitTime: { $ne: null },
      }),

      // Latest 10 parking transactions.
      ParkingTransaction.find()
        .sort({ createdAt: -1 })
        .limit(10)
        .select(
          'transactionId entryTicketNumber vehicleNumberPlate vehicleType slotId entryTime exitTime fee paymentStatus paymentMethod'
        )
        .lean(),

      // Current slot information.
      ParkingSlot.find()
        .sort({ zone: 1, slotNumber: 1 })
        .select('slotId zone slotNumber status vehicleType')
        .lean(),
    ]);

    const occupancyRate =
      totalSlots === 0
        ? 0
        : Number(((occupiedSlots / totalSlots) * 100).toFixed(2));

    return sendSuccess(res, 'Admin dashboard data retrieved.', {
      summary: {
        totalSlots,
        availableSlots,
        occupiedSlots,
        reservedSlots,
        activeSessions,
        completedSessions,
        occupancyRate,
      },
      slots,
      recentTransactions,
    });
  } catch (error) {
    return next(error);
  }
}

module.exports = { getDashboard };

import React, { useCallback, useEffect, useState } from 'react';

export default function AdminDashboard({ apiBaseUrl }) {
  const [dashboard, setDashboard] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [lastUpdated, setLastUpdated] = useState(null);

  const loadDashboard = useCallback(async () => {
    try {
      const response = await fetch(`${apiBaseUrl}/admin/dashboard`, {
        method: 'GET',
        credentials: 'include',
      });

      const result = await response.json();

      if (!response.ok) {
        throw new Error(
          result.message || 'Could not load the Admin dashboard.'
        );
      }

      const data = result.data ?? result;

      setDashboard(data);
      setLastUpdated(new Date());
      setError('');
    } catch (err) {
      setError(
        err instanceof TypeError
          ? 'Cannot connect to the backend. Check that the server is running.'
          : err.message
      );
    } finally {
      setLoading(false);
    }
  }, [apiBaseUrl]);

  useEffect(() => {
    loadDashboard();

    const timer = setInterval(loadDashboard, 10000);

    return () => clearInterval(timer);
  }, [loadDashboard]);

  if (loading && !dashboard) {
    return (
      <section className="admin-dashboard">
        <p className="subtitle">Loading Admin dashboard...</p>
      </section>
    );
  }

  const summary = dashboard?.summary;
  const slots = dashboard?.slots ?? [];
  const transactions = dashboard?.recentTransactions ?? [];

  return (
    <section className="admin-dashboard">
      <div className="admin-dashboard-header">
        <div>
          <p className="admin-eyebrow">ADMINISTRATION</p>
          <h2>Parking Management Dashboard</h2>
          <p className="admin-subtitle">
            Monitor parking operations and saved records.
          </p>
        </div>

        <button
          className="admin-refresh-button"
          type="button"
          onClick={loadDashboard}
          disabled={loading}
        >
          {loading ? 'Refreshing...' : 'Refresh'}
        </button>
      </div>

      {error && (
        <div className="notice failure" role="alert">
          {error}
        </div>
      )}

      {summary && (
        <>
          <div className="admin-stat-grid">
            <div className="admin-stat-card">
              <span>Total slots</span>
              <strong>{summary.totalSlots}</strong>
            </div>

            <div className="admin-stat-card available-stat">
              <span>Available</span>
              <strong>{summary.availableSlots}</strong>
            </div>

            <div className="admin-stat-card occupied-stat">
              <span>Occupied</span>
              <strong>{summary.occupiedSlots}</strong>
            </div>

            <div className="admin-stat-card">

========== server/src/models/ParkingSlot.js ==========
const mongoose = require('mongoose');

const VEHICLE_TYPES = ['Two-wheeler', 'Four-wheeler', 'Other'];
const schema = new mongoose.Schema({
  slotId: { type: String, required: true, unique: true, trim: true, maxlength: 10 },
  zone: { type: String, required: true, trim: true, uppercase: true, maxlength: 10 },
  slotNumber: { type: String, required: true, trim: true, maxlength: 10 },
  status: { type: String, enum: ['Available', 'Occupied', 'Reserved'], required: true, default: 'Available', maxlength: 10 },
  vehicleType: { type: String, enum: VEHICLE_TYPES, required: true, maxlength: 20 }
}, { timestamps: true });
schema.index({ zone: 1, slotNumber: 1 }, { unique: true });
module.exports = mongoose.model('ParkingSlot', schema);

const mongoose = require('mongoose');

const schema = new mongoose.Schema({
  transactionId: { type: String, required: true, unique: true, trim: true, maxlength: 24 },
  vehicleNumberPlate: { type: String, required: true, trim: true, uppercase: true, maxlength: 15 },
  vehicleType: { type: String, enum: ['Two-wheeler', 'Four-wheeler', 'Other'], required: true, maxlength: 20 },
  slotId: { type: String, required: true, trim: true, maxlength: 10 },
  entryTicketNumber: { type: String, required: true, unique: true, trim: true, maxlength: 20 },
  entryTime: { type: Date, required: true, default: Date.now },
  exitTime: { type: Date, default: null },
  fee: { type: Number, min: 0, default: null },
  paymentStatus: { type: String, enum: ['Paid', 'Pending'], default: 'Pending', maxlength: 10 },
  paymentMethod: { type: String, enum: ['Cash', 'Online', null], default: null, maxlength: 10 }
}, { timestamps: true });

// Only one open session (no exit time) per plate; enforced by MongoDB, not app code.
schema.index(
  { vehicleNumberPlate: 1 },
  { unique: true, partialFilterExpression: { exitTime: { $type: 'null' } } }
);

module.exports = mongoose.model('ParkingTransaction', schema);
