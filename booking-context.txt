const jwt = require('jsonwebtoken');
const User = require('../models/User');
const { sendError } = require('../utils/apiResponse');

async function requireAuth(req, res, next) {
  try {
    const token = req.cookies?.parking_token;
    if (!token) return sendError(res, 'Not authenticated.', 401);

    if (!process.env.JWT_SECRET || process.env.JWT_SECRET.length < 32) {
      throw new Error('JWT_SECRET must be at least 32 characters.');
    }

    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    const user = await User.findById(decoded.sub);
    if (!user || !user.isActive) return sendError(res, 'Not authenticated.', 401);

    req.user = user;
    return next();
  } catch (error) {
    if (
      error.name === 'JsonWebTokenError' ||
      error.name === 'TokenExpiredError' ||
      error.name === 'NotBeforeError'
    ) {
      return sendError(res, 'Not authenticated.', 401);
    }
    return next(error);
  }
}

function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return sendError(res, 'You do not have permission to perform this action.', 403);
    }
    return next();
  };
}

module.exports = { requireAuth, requireRole };

  if (err && err.type === 'entity.parse.failed') {
    return sendError(
      res,
      'Request body must be valid JSON.',
      400
    );
  }

  if (err && err.name === 'ValidationError') {
    return sendError(
      res,
      'Submitted data failed validation.',
      400
    );
  }

  if (err && err.code === 11000) {
    return sendError(
      res,
      'A record with that unique value already exists.',
      409
    );
  }

  console.error(err);

  return sendError(res, 'Internal server error.', 500);
});

const HEALTH_URL = `${API_BASE_URL}/health`;

function App() {
  const resetToken = new URLSearchParams(window.location.search).get('token');

  const [mode, setMode] = useState(
    resetToken ? 'reset' : 'login'
  );

  const [showPassword, setShowPassword] = useState(false);
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');

  const [form, setForm] = useState({
    name: '',
    email: '',
    password: '',
    contactNumber: '',
  });

  const [message, setMessage] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const [checkingSession, setCheckingSession] = useState(true);
  const [user, setUser] = useState(null);

  useEffect(() => {
    let cancelled = false;

    async function restoreSession() {
      try {
        const response = await fetch(`${API_URL}/me`, {
          method: 'GET',
          credentials: 'include',
        });

        if (!response.ok) {
          if (response.status !== 401) {
            throw new Error('Could not check your login session.');
          }
          return;
        }

        const result = await response.json();
        const data = result.data ?? result;

        if (!cancelled && data.user) {
          setUser(data.user);
        }
      } catch {
        if (!cancelled) {
          setError(
            'Could not connect to the authentication service. Check that the backend is running.'
          );
        }
      } finally {
        if (!cancelled) {
          setCheckingSession(false);
        }
      }
    }

    restoreSession();

    return () => {
      cancelled = true;
    };
  }, []);

  function updateField(event) {
    const { name, value } = event.target;

    setForm((previous) => ({
      ...previous,
      [name]: value,
    }));

    setError('');
  }

  function switchMode(nextMode) {
    setMode(nextMode);
    setShowPassword(false);
    setPassword('');
    setConfirmPassword('');
    setMessage('');
    setError('');
  }

  async function handleSubmit(event) {
    event.preventDefault();
    setMessage('');
    setError('');

    const isRegister = mode === 'register';

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) {
      setError('Enter a valid email address.');
      return;
    }

    if (isRegister) {
      if (form.name.trim().length < 2 || form.name.trim().length > 50) {
        setError('Name must contain 2 to 50 characters.');
        return;
      }

      if (!form.contactNumber.trim() || form.contactNumber.trim().length > 15) {
        setError('Enter a valid contact number.');
        return;
      }
    }

    if (
      typeof form.password !== 'string' ||
      (isRegister && (form.password.length < 8 || form.password.length > 128))
    ) {
      setError('Password must contain between 8 and 128 characters.');
      return;
    }

    setLoading(true);

    try {
      const endpoint = isRegister ? 'register' : 'login';

      const payload = isRegister
        ? {
            name: form.name.trim(),
            email: form.email.trim(),
            password: form.password,
            contactNumber: form.contactNumber.trim(),
          }
        : {
            email: form.email.trim(),
            password: form.password,
          };

      const response = await fetch(`${API_URL}/${endpoint}`, {
        method: 'POST',
        credentials: 'include',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      const result = await response.json().catch(() => ({}));

      if (!response.ok) {
        throw new Error(result.message || 'The request failed.');
      }

      const data = result.data ?? result;

      if (isRegister) {
        setMessage(
          result.message || 'Registration successful! You can now log in.'
        );

        setForm({
          name: '',
          email: form.email.trim(),
          password: '',
          contactNumber: '',
        });

        setMode('login');
      } else {
        if (!data.user) {
          throw new Error(
            'Login response did not contain user information.'
          );
        }

        setUser(data.user);

        setForm((previous) => ({
          ...previous,
          password: '',
        }));

        setMessage('You have logged in successfully.');
      }
    } catch (err) {
      setError(
        err instanceof TypeError
          ? 'Cannot reach the API. Check that the backend is running.'
          : err.message || 'Something went wrong.'
      );
    } finally {
      setLoading(false);
    }
  }

    event.preventDefault();
    setMessage('');
    setError('');

          'Content-Type': 'application/json',

        body: JSON.stringify({

        throw new Error(result.message || 'Could not process your request.');

      setMessage(

      );
    } catch (err) {
      setError(
        err instanceof TypeError
          ? 'Cannot reach the API. Check that the backend is running.'
          : err.message || 'Something went wrong.'
      );
    } finally {
      setLoading(false);
    }
  }

  async function handleResetPassword(event) {
    event.preventDefault();
    setMessage('');
    setError('');

    if (!resetToken) {
      setError('Password reset link is missing. Request a new link.');
      return;
    }

    if (password.length < 8 || password.length > 128) {
      setError('Password must contain between 8 and 128 characters.');
      return;
    }

    if (password !== confirmPassword) {
      setError('Passwords do not match.');
      return;
    }

    setLoading(true);

    try {
      const response = await fetch(`${API_URL}/reset-password`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          token: resetToken,
          password,
        }),
      });

      const result = await response.json().catch(() => ({}));

      if (!response.ok) {
        throw new Error(result.message || 'Password reset failed.');
      }

      setPassword('');
      setConfirmPassword('');
      setForm((previous) => ({
        ...previous,
        password: '',
      }));

      window.history.replaceState({}, document.title, window.location.pathname);

      setMode('login');
      setMessage(
        result.message || 'Password reset successful. You can now log in.'
      );
    } catch (err) {
      setError(
        err instanceof TypeError
          ? 'Cannot reach the API. Check that the backend is running.'
          : err.message || 'Password reset failed.'
      );

      setLoading(false);

  if (checkingSession) {

import React, { useCallback, useEffect, useState } from 'react';

const VEHICLE_TYPES = ['Two-wheeler', 'Four-wheeler', 'Other'];
const REFRESH_MS = 10000;

export default function SlotAvailability({ apiBaseUrl }) {
  const [vehicleType, setVehicleType] = useState('');
  const [statusFilter, setStatusFilter] = useState('');
  const [data, setData] = useState(null);
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(true);

  const load = useCallback(async () => {
    const params = new URLSearchParams();
    if (vehicleType) params.set('vehicleType', vehicleType);
    if (statusFilter) params.set('status', statusFilter);
    const query = params.toString();

    try {
      const response = await fetch(
        `${apiBaseUrl}/slots${query ? `?${query}` : ''}`,
        { credentials: 'include' }
      );
      const result = await response.json();

      if (!response.ok) {
        throw new Error(result.message || 'Slot availability is currently unavailable.');
      }

      setData(result.data);
      setError('');
    } catch (err) {
      // Keep the last good data on screen, but tell the user it may be out of date.
      setError(
        err instanceof TypeError
          ? 'Slot availability is currently unavailable. Check that the backend is running.'
          : err.message
      );
    } finally {
      setLoading(false);
    }
  }, [apiBaseUrl, vehicleType, statusFilter]);

  // Load now, then poll so the view follows vehicle entries and exits.
  useEffect(() => {
    setLoading(true);
    load();
    const timer = setInterval(load, REFRESH_MS);
    return () => clearInterval(timer);
  }, [load]);

  const summary = data?.summary;

  return (
    <div className="slot-panel">
      <h2>Parking slot availability</h2>

      {error && (
        <div className="notice failure" role="alert">
          {error}
          {data && ' The slots shown below may be out of date.'}
        </div>
      )}

      {summary && (
        <div className="slot-summary" aria-label="Slot counts">
          <span className="slot-count available">Available: {summary.available}</span>
          <span className="slot-count occupied">Occupied: {summary.occupied}</span>
          {summary.reserved > 0 && (
            <span className="slot-count reserved">Reserved: {summary.reserved}</span>
          )}
          <span className="slot-count">Total: {summary.total}</span>
        </div>
      )}

      <div className="slot-filters">
        <select
          aria-label="Filter by vehicle type"
          value={vehicleType}
          onChange={(e) => setVehicleType(e.target.value)}
        >
          <option value="">All vehicle types</option>
          {VEHICLE_TYPES.map((type) => (
            <option key={type} value={type}>{type}</option>
          ))}
        </select>

        <select
          aria-label="Filter by status"
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value)}
        >
          <option value="">All statuses</option>
          <option value="Available">Available</option>
          <option value="Occupied">Occupied</option>
          <option value="Reserved">Reserved</option>
        </select>
      </div>

      {loading && !data && <p className="subtitle">Loading slot availability...</p>}

      {data && data.slots.length === 0 && (
        <p className="subtitle">No parking slots match your search.</p>
      )}

      {data && data.slots.length > 0 && (
        <ul className="slot-grid">
          {data.slots.map((slot) => (
            <li
              key={slot.slotId}
              className={`slot-tile ${slot.status.toLowerCase()}`}
              title={`${slot.slotId} Â· ${slot.vehicleType} Â· ${slot.status}`}
            >
              <strong>{slot.slotId}</strong>
              <span>{slot.status}</span>
              <small>{slot.vehicleType}</small>
            </li>
          ))}
        </ul>
      )}

      <div className="slot-footer">
        {data && (
          <small>Last updated {new Date(data.updatedAt).toLocaleTimeString()}</small>
        )}
        <button className="text-button" type="button" onClick={load}>
          Refresh now
        </button>
      </div>
    </div>
  );
}
